# Keep only features present in the object's active (or named) assay, so DotPlot
# / FeaturePlot never error on a gene absent from the panel. Marker vectors
# below deliberately include aliases (e.g. "Cd11b", "NKp46") that are not row
# names; those are dropped here.
present_features <- function(sobj, features, assay = NULL) {
  feats <- unique(features)
  if (is.null(assay)) assay <- DefaultAssay(sobj)
  avail <- rownames(sobj[[assay]])
  keep <- feats[feats %in% avail]
  missing <- setdiff(feats, keep)
  if (length(missing))
    message("Dropping genes not in assay '", assay, "': ",
            paste(missing, collapse = ", "))
  keep
}

# Compact DotPlot of `features` across the levels of `group_var`, in the shared
# theme used throughout (classic theme, rotated axis, no axis titles). `flip`
# puts genes on the y-axis (rows) for tall marker panels.
cluster_dotplot <- function(sobj, features, group_var, flip = TRUE,
                            title = NULL, cols = c("gray89", "tomato3")) {
  feats <- present_features(sobj, features)
  p <- DotPlot(sobj, features = feats, group.by = group_var, cols = cols) +
    theme_classic() +
    RotatedAxis() +
    xlab("") + ylab("")
  if (flip) p <- p + coord_flip()
  if (!is.null(title)) p <- p + ggtitle(title)
  p
}

# Full re-process of one Seurat cluster: subset, then re-run
# FindVariableFeatures / ScaleData / PCA / UMAP / neighbors / clustering from
# scratch on the sketch assay, producing a fresh "umap.sketch" reduction and a
# `subcluster` label (parentCluster_subID). Heavier than reclustering on the
# existing embedding, but gives clean, cluster-specific subclusters.
recluster_subset <- function(sobj, cl, cluster_col = seurat_cluster_var,
                             assay = recluster_assay, npcs = 30, dims = 1:30,
                             res = 0.5) {
  cells <- colnames(sobj)[!is.na(sobj@meta.data[[cluster_col]]) &
                            sobj@meta.data[[cluster_col]] == cl]
  sub <- subset(sobj, cells = cells)

  DefaultAssay(sub) <- if (assay %in% Assays(sub)) assay else DefaultAssay(sub)

  sub <- FindVariableFeatures(sub, verbose = FALSE)
  sub <- ScaleData(sub, verbose = FALSE)
  sub <- RunPCA(sub, npcs = npcs, reduction.name = "pca.sub", verbose = FALSE)
  sub <- FindNeighbors(sub, reduction = "pca.sub", dims = dims, verbose = FALSE)
  sub <- FindClusters(sub, resolution = res, verbose = FALSE)
  sub <- RunUMAP(sub, reduction = "pca.sub", dims = dims,
                 reduction.name = "umap.sketch",
                 reduction.key = "umapsketch_", verbose = FALSE)

  sub$subcluster <- paste0(cl, "_", as.character(sub$seurat_clusters))
  sub$subcluster <- factor(sub$subcluster,
                           levels = paste0(cl, "_",
                                           sort(unique(as.integer(as.character(sub$seurat_clusters))))))
  sub
}

Reclustering each Seurat cluster and building the working cluster label

Each Seurat cluster (projected_snn_res.1.2) is subset and fully re-processed from scratch (FindVariableFeatures → ScaleData → PCA → UMAP → neighbors → clustering) on the sketch assay, producing a fresh umap.sketch embedding and a subcluster label. Cluster 2 is reclustered at a higher resolution than the other clusters. Selected subclusters are then split out into their own clusters while every other spot keeps its projected_snn_res.1.2 label, giving a new_cluster label. From this point on new_cluster is the primary Seurat cluster identity, used for annotation, marker discovery and the TACCO / predicted-ID comparisons below.
